cmake_minimum_required(VERSION 3.26)
project(sppy LANGUAGES CXX)
find_package(Python 3.11 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 3 CONFIG REQUIRED)
include(cmake/FindSpPyEigen.cmake)
set(_openmp_default OFF)
if(DEFINED ENV{SPPY_OPENMP})
    if("$ENV{SPPY_OPENMP}" STREQUAL "1")
        set(_openmp_default ON)
    endif()
    # The build directory is reused: changing the environment must also
    # override an earlier cached configuration.
    set(SPPY_OPENMP ${_openmp_default} CACHE BOOL "Enable OpenMP in Eigen kernels" FORCE)
endif()
option(SPPY_OPENMP "Enable OpenMP in Eigen kernels" ${_openmp_default})
if(SPPY_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS CXX)
endif()
file(GLOB _sources CONFIGURE_DEPENDS "sppy/native/*.cpp")
pybind11_add_module(_native MODULE NO_EXTRAS ${_sources})
target_compile_features(_native PRIVATE cxx_std_17)
target_link_libraries(_native PRIVATE Eigen3::Eigen)
# Eigen solver templates are memory intensive; serialize compilation even when
# callers request a parallel build. This also makes wheel builds predictable.
set_property(GLOBAL PROPERTY JOB_POOLS eigen_compile=1)
set_property(TARGET _native PROPERTY JOB_POOL_COMPILE eigen_compile)
if(SPPY_OPENMP)
    target_link_libraries(_native PRIVATE OpenMP::OpenMP_CXX)
endif()
install(TARGETS _native LIBRARY DESTINATION sppy RUNTIME DESTINATION sppy)
